#!/usr/bin/env python3
import sys, os
from pathlib import Path
from xml.dom import minidom

def pascal_case(s):
    return s.replace('-', ' ').title().replace(' ', '')

def main():
    if len(sys.argv) < 2:
        print("Usage: svg_to_ng.py input.svg [output_dir]")
        sys.exit(1)

    input_file = Path(sys.argv[1])
    output_dir = Path(sys.argv[2]) if len(sys.argv) > 2 else Path("src/app/icons")

    name = input_file.stem
    class_name = "Icon" + pascal_case(name)
    selector = f"icon-{name}"

    with open(input_file, "r", encoding="utf-8") as f:
        svg_data = f.read()

    os.makedirs(output_dir, exist_ok=True)
    output_file = output_dir / f"{name}.ts"

    template = f"""
import {{ Component }} from '@angular/core';

@Component({{
  selector: '{selector}',
  standalone: true,
  template: `{svg_data}`,
}})
export class {class_name} {{}}
"""

    with open(output_file, "w", encoding="utf-8") as f:
        f.write(template.strip())

    print(f"✅ Created {output_file.relative_to(Path.cwd())}")

if __name__ == "__main__":
    main()
